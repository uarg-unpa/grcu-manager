@startuml
actor "Usuario autorizado" as Usuario
participant "Interfaz de usuario" as UI
participant "Gestor de Casos de Uso" as Gestor
database "BDD" as BDD
collections "Caché" as Cache

== Inicio ==
Usuario -> UI : Acceder al caso de uso
UI -> Gestor : obtenerDetalle(cuId)
Gestor -> BDD : buscarCasoDeUso(cuId)
BDD --> Gestor : datosCU
Gestor -> UI : mostrarDetalle(datosCU)

== Comentario ==
Usuario -> UI : Seleccionar pestaña "Comentarios"
UI -> Gestor : habilitarCampoTexto()
Usuario -> Gestor : Enviar comentario(texto)

alt Comentario vacío
  Gestor -> UI : mostrarError("El comentario no puede estar vacío")
else Comentario con texto
  Gestor -> Gestor : validarLongitud(texto)

  alt Excede límite de caracteres
    Gestor -> UI : notificar("El comentario excede el límite permitido")
  else Longitud válida
    Gestor -> BDD : registrarComentarioCU(cuId, usuarioId, texto, fecha)
    BDD --> Gestor : {OK / ERROR}

    alt Guardado OK
      Gestor -> UI : actualizarHiloComentarios()
      UI -> Usuario : Mostrar nuevo comentario
    else Falla en la carga
      Gestor -> Cache : guardarTemporal(cuId, texto, fecha)
      Cache --> Gestor : OK
      Gestor -> UI : permitirReintento("Error al guardar, intente nuevamente")
    end
  end
end
@enduml
