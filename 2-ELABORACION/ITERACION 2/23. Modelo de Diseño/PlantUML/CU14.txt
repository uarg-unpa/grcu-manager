@startuml
actor "Usuario autorizado" as Usuario
participant "Interfaz de usuario" as UI
participant "Gestor de Casos de Uso" as Gestor
database "BDD" as BDD
collections "Almacenamiento / Enlaces" as Repo
collections "Caché" as Cache

== Inicio ==
Usuario -> UI : Abrir caso de uso
UI -> Gestor : obtenerDetalle(cuId)
Gestor -> BDD : buscarCasoDeUso(cuId)
BDD --> Gestor : datosCU
Gestor -> UI : mostrarDetalle(datosCU)

== Adjuntar archivo ==
Usuario -> UI : Seleccionar "Adjuntar archivo"
UI -> Gestor : abrirDialogoCarga()
Gestor -> Usuario : Mostrar opciones (archivo o enlace)
Usuario -> Gestor : Enviar archivo o enlace

== Validación ==
Gestor -> Gestor : validarFormato(archivo/enlace)
Gestor -> Gestor : validarTamano(archivo)

alt Formato no permitido
  Gestor -> UI : mostrarError("Formato no válido")
else Tamaño excedido
  Gestor -> UI : notificar("Archivo supera el tamaño permitido")
else Archivo válido
  Gestor -> Repo : guardarArchivo(cuId, archivo/enlace)
  Repo --> Gestor : {OK / ERROR}

  alt Guardado OK
    Gestor -> BDD : registrarReferenciaArchivoCU(cuId, ruta/enlace)
    BDD --> Gestor : OK
    Gestor -> BDD : asegurarPermisosLecturaParaMiembros(cuId)
    BDD --> Gestor : OK
    Gestor -> UI : actualizarVistaConArchivo()
    UI -> Usuario : Archivo disponible para miembros con lectura
  else Falla de carga
    Gestor -> Cache : guardarIntentoTemporal(cuId, metadatos)
    Cache --> Gestor : OK
    Gestor -> UI : permitirReintento("Fallo de carga, intente nuevamente")
  end
end
@enduml
